22. 自动采集：底层原始 1920x1536 直出 fps
-----------------------------------------
bash src/usv_perception/usv_vision/scripts/jetson_collect_benchmarks.sh \
  --case-name raw_1920_direct_front \
  --mode gst \
  --gst-cmd "gst-launch-1.0 -ev v4l2src device=/dev/video4 ! video/x-raw,format=UYVY,width=1920,height=1536,framerate=30/1 ! fpsdisplaysink text-overlay=1 video-sink=fakesink sync=0" \
  --duration 20 \
  --device-or-topic /dev/video4 \
  --pipeline-type raw_direct \
  --target-resolution 1920x1536 \
  --results-dir /home/cczh/USV_ROS/test_logs/benchmarks \
  --notes "front camera raw direct baseline"
